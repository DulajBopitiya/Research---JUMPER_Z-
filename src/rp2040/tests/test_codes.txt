#include <Arduino.h>
#include <Adafruit_NeoPixel.h>

// ====== SETTINGS ======
#define LED_PIN      25        // WS2812 data pin (GPIO25)
#define LED_COUNT    400        // <-- set your LED count
#define BRIGHTNESS   10        // 0..255 (keep modest for testing)

Adafruit_NeoPixel strip(LED_COUNT, LED_PIN, NEO_GRB + NEO_KHZ800);

// ----------------- helpers -----------------
static inline uint32_t rgb(uint8_t r, uint8_t g, uint8_t b)
{
  return strip.Color(r, g, b);
}

static void clearStrip()
{
  strip.clear();
  strip.show();
}

// Fade all pixels by 'amount' (0..255). Bigger amount = faster fade.
static void fadeAll(uint8_t amount)
{
  for (uint16_t i = 0; i < strip.numPixels(); i++)
  {
    uint32_t c = strip.getPixelColor(i);
    uint8_t r = (uint8_t)(c >> 16);
    uint8_t g = (uint8_t)(c >> 8);
    uint8_t b = (uint8_t)(c);

    r = (r * (255 - amount)) / 255;
    g = (g * (255 - amount)) / 255;
    b = (b * (255 - amount)) / 255;

    strip.setPixelColor(i, r, g, b);
  }
}

// ----------------- effects -----------------

static void colorWipe(uint32_t color, uint16_t waitMs)
{
  for (uint16_t i = 0; i < strip.numPixels(); i++)
  {
    strip.setPixelColor(i, color);
    strip.show();
    delay(waitMs);
  }
}

static void rainbowCycle(uint16_t waitMs)
{
  for (uint16_t j = 0; j < 256; j++)
  {
    for (uint16_t i = 0; i < strip.numPixels(); i++)
    {
      uint8_t pos = (i * 256 / strip.numPixels() + j) & 0xFF;

      uint8_t r, g, b;
      if (pos < 85) { r = pos * 3; g = 255 - pos * 3; b = 0; }
      else if (pos < 170) { pos -= 85; r = 255 - pos * 3; g = 0; b = pos * 3; }
      else { pos -= 170; r = 0; g = pos * 3; b = 255 - pos * 3; }

      strip.setPixelColor(i, r, g, b);
    }
    strip.show();
    delay(waitMs);
  }
}

// Theater chase: moving dotted line
static void theaterChase(uint32_t color, uint16_t waitMs, uint8_t cycles)
{
  for (uint8_t c = 0; c < cycles; c++)
  {
    for (uint8_t q = 0; q < 3; q++)
    {
      strip.clear();
      for (uint16_t i = q; i < strip.numPixels(); i += 3)
        strip.setPixelColor(i, color);

      strip.show();
      delay(waitMs);
    }
  }
}

// Breathing brightness pulse (whole strip)
static void breathe(uint32_t color, uint16_t stepDelayMs, uint8_t cycles)
{
  for (uint8_t k = 0; k < cycles; k++)
  {
    for (int b = 0; b <= 255; b += 5)
    {
      strip.setBrightness((uint8_t)((BRIGHTNESS * b) / 255));
      for (uint16_t i = 0; i < strip.numPixels(); i++)
        strip.setPixelColor(i, color);

      strip.show();
      delay(stepDelayMs);
    }

    for (int b = 255; b >= 0; b -= 5)
    {
      strip.setBrightness((uint8_t)((BRIGHTNESS * b) / 255));
      for (uint16_t i = 0; i < strip.numPixels(); i++)
        strip.setPixelColor(i, color);

      strip.show();
      delay(stepDelayMs);
    }
  }
  strip.setBrightness(BRIGHTNESS);
}

// Scanner (Larson/Cylon): dot back and forth with fade trail
static void scanner(uint32_t color, uint8_t trailFade, uint16_t stepDelayMs, uint8_t cycles)
{
  for (uint8_t c = 0; c < cycles; c++)
  {
    for (int i = 0; i < (int)strip.numPixels(); i++)
    {
      fadeAll(trailFade);
      strip.setPixelColor(i, color);
      strip.show();
      delay(stepDelayMs);
    }
    for (int i = (int)strip.numPixels() - 1; i >= 0; i--)
    {
      fadeAll(trailFade);
      strip.setPixelColor(i, color);
      strip.show();
      delay(stepDelayMs);
    }
  }
}

// Comet / meteor: bright head with fading tail
static void comet(uint32_t headColor, uint8_t tailFade, uint16_t stepDelayMs, uint8_t rounds)
{
  for (uint8_t r = 0; r < rounds; r++)
  {
    clearStrip();
    for (int pos = 0; pos < (int)strip.numPixels() + 10; pos++)
    {
      fadeAll(tailFade);
      if (pos >= 0 && pos < (int)strip.numPixels())
        strip.setPixelColor(pos, headColor);

      strip.show();
      delay(stepDelayMs);
    }
  }
}

// Sparkle: random twinkles
static void sparkle(uint32_t color, uint16_t runtimeMs, uint16_t sparkleDelayMs)
{
  uint32_t start = millis();
  clearStrip();

  while ((millis() - start) < runtimeMs)
  {
    fadeAll(40); // gentle fade

    int p = random((int)strip.numPixels()); // RP2040-safe
    strip.setPixelColor(p, color);

    strip.show();
    delay(sparkleDelayMs);
  }
}

// ----------------- main -----------------
void setup()
{
  Serial.begin(115200);
  delay(200);

  // RP2040-safe random seed (no esp_random)
  randomSeed((uint32_t)micros() ^ (uint32_t)millis());

  strip.begin();
  strip.setBrightness(BRIGHTNESS);
  strip.show();

  Serial.println("WS2812B effects demo (RP2040, GPIO25)");
}

void loop()
{
  // Basic wipes     c    
  colorWipe(rgb(255, 0, 0), 15);
  colorWipe(rgb(0, 255, 0), 15);
  colorWipe(rgb(0, 0, 255), 15);
  colorWipe(rgb(0, 0, 0), 10);

  // Rainbow
  rainbowCycle(4);

  // Theater chase (warm amber)
  theaterChase(rgb(255, 180, 0), 70, 25);

  // Breathing pulse (cyan)
  breathe(rgb(0, 150, 255), 10, 2);

  // Scanner (red)
  scanner(rgb(255, 0, 0), 50, 12, 4);

  // Comet (green-ish)
  comet(rgb(0, 255, 120), 55, 15, 3);

  // Sparkle (white twinkle)
  sparkle(rgb(255, 255, 255), 3000, 30);

  clearStrip();
  delay(200);
}

  JsonBridge::begin(LedMatrix::strip());
}

// ==========================================================
// ========================= LOOP ============================
// ==========================================================

void loop()
{

  if (USB_CDC_Config::USBSer1.available())
  {
    JsonDocument req;
    DeserializationError err = deserializeJson(req, USB_CDC_Config::USBSer1);

    if (!err)
    {
      JsonBridge::handle(USB_CDC_Config::USBSer1, req);
    }
  }

  JsonBridge::tick();
}

#include "configuration.h"


// pulse helpers
static inline void clkPulse() {
  digitalWrite(CH_CLK, HIGH);
  delayMicroseconds(2);
  digitalWrite(CH_CLK, LOW);
  delayMicroseconds(2);
}

static inline void stbPulse() {
  digitalWrite(STB_A, HIGH);
  delayMicroseconds(5);
  digitalWrite(STB_A, LOW);
  delayMicroseconds(5);
}

// Shift 7-bit address MSB->LSB on DAT, clocked by CH_CLK rising edges
static void ch_shiftAddress7(uint8_t addr7)
{
  for (int i = 6; i >= 0; i--) {
    digitalWrite(CH_DATA, (addr7 >> i) & 0x01);
    // datasheet says rising edge is valid in serial mode
    clkPulse();
  }
}

// Write one crosspoint (X0..15, Y0..7), on=true/false
static void ch_writeSwitch(uint8_t x, uint8_t y, bool on)
{
  if (x > 15 || y > 7) return;

  uint8_t addr = (y << 4) | x; // addressing table in datasheet

  // 1) shift address
  ch_shiftAddress7(addr);

  // 2) put data bit on DAT (1=ON, 0=OFF)
  digitalWrite(CH_DATA, on ? HIGH : LOW);
  delayMicroseconds(2);

  // 3) STB high pulse to execute/latch
  stbPulse();

  // leave DAT low
  digitalWrite(CH_DATA, LOW);
}

static void ch_reset()
{
  // RST is ACTIVE HIGH: high clears all latches (all switches open)
  digitalWrite(CH_RST, HIGH);
  delay(10);
  digitalWrite(CH_RST, LOW);
  delay(10);
}

void setup()
{
  Serial.begin(115200);

  pinMode(CH_DATA, OUTPUT);
  pinMode(CH_CLK, OUTPUT);
  pinMode(CH_RST, OUTPUT);
  pinMode(STB_A, OUTPUT);

  digitalWrite(CH_DATA, LOW);
  digitalWrite(CH_CLK, LOW);
  digitalWrite(STB_A, LOW);
  digitalWrite(CH_RST, LOW);

  Serial.println("CH446Q SERIAL MODE TEST");

  ch_reset();
  delay(500);


}
